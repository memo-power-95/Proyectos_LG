===============================================
GUÍA COMPLETA: COMANDOS CMD Y POWERSHELL
===============================================

NAVEGACIÓN Y DIRECTORIOS
===============================================

CMD:
  cd                        Ver carpeta actual
  cd carpeta                Entrar a una carpeta
  cd ..                     Subir una carpeta
  cd \                      Ir a raíz del disco
  cd /d D:\ruta             Cambiar de disco y carpeta
  D:                        Cambiar a disco D
  dir                       Listar archivos y carpetas
  dir /S                    Listar recursivo (incluyendo subcarpetas)
  dir /B                    Listar solo nombres
  tree                      Ver estructura en árbol
  tree /F                   Ver estructura con archivos

POWERSHELL:
  Get-Location              Ver carpeta actual (alias: pwd)
  Set-Location C:\ruta      Cambiar de carpeta (alias: cd)
  Get-ChildItem             Listar contenido (alias: ls, dir)
  Get-ChildItem -Recurse    Listar recursivo
  Push-Location             Guardar ubicación actual
  Pop-Location              Volver a ubicación anterior

===============================================
VER Y BUSCAR ARCHIVOS
===============================================

CMD:
  dir *.csv                 Buscar archivos CSV
  dir /S *.csv              Buscar CSV en todas las carpetas
  findstr /R "texto" *.txt  Buscar texto en archivos
  where archivo.exe         Buscar archivo en PATH
  dir C:\ /S               Listar todo disco C

POWERSHELL:
  Get-ChildItem -Filter "*.csv"
                            Buscar archivos CSV
  Get-ChildItem -Recurse -Filter "*.csv"
                            Buscar CSV recursivo
  Get-ChildItem -Filter "*.txt" | Select-Object Name, Length
                            Listar TXT con tamaño
  Search-String "texto" *.txt
                            Buscar texto en archivos

===============================================
ELIMINAR ARCHIVOS Y CARPETAS
===============================================

CMD:
  del archivo.txt           Eliminar archivo
  del *.csv                 Eliminar todos los CSV
  del /P *.csv              Eliminar con confirmación
  del /Q *.csv              Eliminar sin confirmación
  del /S *.csv              Eliminar en subcarpetas
  rmdir carpeta             Eliminar carpeta vacía
  rmdir /S carpeta          Eliminar carpeta y contenido
  rmdir /S /Q carpeta       Eliminar sin confirmación
  forfiles /M "*.csv" /D -30 /C "cmd /c del @file"
                            Eliminar archivos de últimos 30 días

POWERSHELL:
  Remove-Item archivo.txt   Eliminar archivo
  Remove-Item "*.csv"       Eliminar CSV
  Remove-Item -Recurse carpeta
                            Eliminar carpeta y contenido
  Remove-Item -Force archivo.txt
                            Forzar eliminación
  Get-ChildItem "*.csv" | Remove-Item
                            Pipeline: listar y eliminar

                            Renombrar
  Get-Content archivo.txt   Ver contenido
  Set-Content archivo.txt "texto"
                            Crear/escribir archivo

===============================================
VER TAMAÑO DE ARCHIVOS Y DISCO
===============================================

CMD:
  dir "C:\ruta" /S          Ver tamaño de carpeta
  tree "C:\ruta" /A /S      Ver estructura con tamaño
  fsutil volume diskfree C: Ver espacio libre disco C

POWERSHELL:
  Get-ChildItem -Recurse | Measure-Object -Property Length -Sum
                            Ver tamaño total carpeta
  Get-ChildItem "C:\" -Directory | ForEach-Object {$size = (Get-ChildItem $_.FullName -Recurse -ErrorAction SilentlyContinue | Measure-Object -Property Length -Sum).Sum; [PSCustomObject]@{Carpeta=$_.Name; "Tamaño (GB)"=[math]::Round($size/1GB,2)}} | Sort-Object "Tamaño (GB)" -Descending | Select-Object -First 20
                            Top 20 carpetas por tamaño
  Get-ChildItem -Recurse -File | Sort-Object Length -Descending | Select-Object -First 10 FullName, @{Name="Tamaño (MB)"; Expression={[math]::Round($_.Length/1MB,2)}}
                            Top 10 archivos más grandes
  Get-Volume | Select-Object DriveLetter, @{Name="Total (GB)"; Expression={[math]::Round($_.Size/1GB,2)}}, @{Name="Libre (GB)"; Expression={[math]::Round($_.SizeRemaining/1GB,2)}}
                            Espacio disco resumido

VARIABLES Y SCRIPTING
===============================================

CMD:
  set VARIABLE=valor        Crear variable
  echo %VARIABLE%           Ver variable
  set                       Ver todas las variables
  if %VARIABLE%==valor (comando)
                            Condición IF
  for %i in (1,2,3) do echo %i
                            Loop FOR

POWERSHELL:
  $variable = "valor"       Crear variable
  $variable                 Ver variable
  Get-Variable              Ver todas las variables
  if ($variable -eq "valor") { comando }
                            Condición IF
  foreach ($i in 1..10) { $i }
                            Loop FOR
  1..10 | ForEach-Object { $_ }
                            Loop con pipeline

===============================================
COMBINACIONES ÚTILES (PIPELINES POWERSHELL)
===============================================

Ver archivos CSV grandes:
  Get-ChildItem -Recurse -Filter "*.csv" | Sort-Object Length -Descending | Select-Object -First 5 FullName, @{Name="Tamaño (MB)"; Expression={[math]::Round($_.Length/1MB,2)}}

Contar archivos por tipo:
  Get-ChildItem -Recurse | Group-Object Extension | Select-Object Name, Count

Eliminar archivos mayores a 100MB:
  Get-ChildItem -Recurse -File | Where-Object {$_.Length -gt 100MB} | Remove-Item

Buscar archivos modificados hoy:
  Get-ChildItem -Recurse | Where-Object {$_.LastWriteTime -gt (Get-Date).Date}

Crear lista de archivos en TXT:
  Get-ChildItem -Recurse | Select-Object FullName | Export-Csv "lista.csv" -NoTypeInformation
